sha256 = hashlib.sha256(path.read_bytes()).hexdigest()
normalized = re.sub(r"[\s\u200b-\u200d\ufeff]", "", unicodedata.normalize("NFKC", text).lower())
group = hashlib.sha256(normalized.encode("utf-8")).hexdigest()
bucket = int(group[:8], 16) % 20
split = "development" if bucket < 14 else "calibration" if bucket < 17 else "test"
